export const DEVICE_TYPE_LIST = ['desktop', 'mobile'] as const;
export const deviceTypeSchema = z.enum(DEVICE_TYPE_LIST);
export type DeviceType = z.infer<typeof deviceTypeSchema>;

export const DEVICE_ID_SEPARATOR = ':';

// Un deviceId es `<deviceType>:<value>` con un value no vacÃ­o.
export const deviceIdSchema = z
  .string()
  .min(3)
  .max(128)
  .refine(
    (candidateDeviceId) => {
      const separatorIndex = candidateDeviceId.indexOf(DEVICE_ID_SEPARATOR);
      if (separatorIndex <= 0) {
        return false;
      }
      const prefix = candidateDeviceId.slice(0, separatorIndex);
      const value = candidateDeviceId.slice(separatorIndex + 1);
      return (
        (DEVICE_TYPE_LIST as readonly string[]).includes(prefix) &&
        value.trim().length > 0
      );
    },
    {
      message: 'deviceId invÃ¡lido: se espera "<desktop|mobile>:<valor>"',
    },
  );

// El prefijo del deviceId tiene que concordar con el deviceType declarado: si no,
// un cliente podrÃ­a pedir targeting por tipo y recibir el device equivocado.
export function parseDeviceIdentity(input: {
  readonly deviceId: string;
  readonly deviceType: DeviceType;
  readonly identitySource: DeviceIdentitySource;
}): DeviceIdentity | DeviceIdentityRejection {
  const parsedDeviceId = deviceIdSchema.safeParse(input.deviceId);
  if (!parsedDeviceId.success) {
    return { reason: 'malformed_device_id', detail: input.deviceId };
  }
  const separatorIndex = input.deviceId.indexOf(DEVICE_ID_SEPARATOR);
  const idDeviceType = input.deviceId.slice(0, separatorIndex);
  if (idDeviceType !== input.deviceType) {
    return {
      reason: 'device_id_type_mismatch',
      detail: `deviceId dice ${idDeviceType} pero se declarÃ³ ${input.deviceType}`,
    };
  }
  // Un mobile con identidad heredada del hostname serÃ­a frÃ¡gil: el telÃ©fono
  // cambia de hostname y romperÃ­a el registro.
  if (input.deviceType === 'mobile' && input.identitySource !== 'client_provided') {
    return {
      reason: 'mobile_requires_client_provided_identity',
      detail: 'mobile no puede registrarse con identidad derivada de hostname',
    };
  }
  return {
    deviceId: input.deviceId,
    deviceType: input.deviceType,
    identitySource: input.identitySource,
  };
}

export function isDeviceIdentityRejection(
  value: DeviceIdentity | DeviceIdentityRejection,

  return 'reason' in value;
}

function encodeBase64Url(byteList: Uint8Array): string {
  let binary = '';
  for (const byte of byteList) {
    binary += String.fromCharCode(byte);
  }
  return btoa(binary).replaceAll('+', '-').replaceAll('/', '_').replaceAll('=', '');
}

function readRandomBase64Url(byteCount: number): string {
  const randomByteList = new Uint8Array(byteCount);
  crypto.getRandomValues(randomByteList);
  return encodeBase64Url(randomByteList);
}

/**
 * SHA-256 en hex. WebCrypto es asÃ­ncrono y no tiene streaming, pero un secreto
 * de 43 caracteres entra de sobra: no hace falta HKDF ni nada derivado.

export async function hashSecret(plainSecret: string): Promise<string> {
  const digest = await crypto.subtle.digest(
    'SHA-256',
    new TextEncoder().encode(plainSecret),
  );
  return [...new Uint8Array(digest)]
    .map((byte) => byte.toString(16).padStart(2, '0'))
    .join('');
}

export async function verifyDeviceCredential(input: {

  readonly credential: string | null;

  }

  for (const record of await input.store.listCredentialRecordList()) {
    if (!areHashesEqualTimingSafe(presentedHash, record.credentialHash)) {
      continue;
    }

function areHashesEqualTimingSafe(leftHash: string, rightHash: string): boolean {
  if (leftHash.length !== rightHash.length) {
    return false;
  }
  let difference = 0;
  for (let index = 0; index < leftHash.length; index += 1) {
    difference |= leftHash.charCodeAt(index) ^ rightHash.charCodeAt(index);
  }
  return difference === 0;
}
