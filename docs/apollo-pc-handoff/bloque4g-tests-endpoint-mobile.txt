class FakeWebSocket {
  readonly sentMessageList: string[] = [];
  closeCallCount = 0;

  send(messageText: string): void {
    this.sentMessageList.push(messageText);
  }

  close(): void {
    this.closeCallCount += 1;
  }

  listParsedMessageList(): Record<string, unknown>[] {
    return this.sentMessageList.map(
      (messageText) => JSON.parse(messageText) as Record<string, unknown>,
    );
  }
}

class CountingTimerPort implements HeartbeatTimerPort {
  #nextHandle = 1;
  readonly #activeHandleSet = new Set<number>();
  createdCount = 0;

  setInterval(): HeartbeatTimerHandle {
    this.createdCount += 1;
    const handle = this.#nextHandle;
    this.#nextHandle += 1;
    this.#activeHandleSet.add(handle);
    return handle;
  }

  clearInterval(handle: HeartbeatTimerHandle): void {
    this.#activeHandleSet.delete(handle as number);
  }

  get activeCount(): number {
    return this.#activeHandleSet.size;
  }
}

type Harness = {
  readonly endpoint: MobileDeviceEndpoint;
  readonly timerPort: CountingTimerPort;
  readonly setNow: (value: number) => void;
};

function buildEndpoint(now: number = START): Harness {
  let current = now;
  const timerPort = new CountingTimerPort();
  const endpoint = new MobileDeviceEndpoint({
    nowMilliseconds: () => current,
    timerPort,
  });
  return { endpoint, timerPort, setNow: (value: number) => (current = value) };
}

// El request real que harÃ­a el telÃ©fono, pasado por la decisiÃ³n de ruta y por el
// lÃ­mite de autenticaciÃ³n, antes de tocar el endpoint.
async function authorizeUpgrade(
  pathname: string,
  token: string | null,
): Promise<Response | undefined> {
  const routeDecision = resolveMobileBridgeRoute({
    pathname,
    upgradeHeader: 'websocket',
  });
  if (!routeDecision.isMobileBridgeRoute) {
    return new Response('Not a mobile route', { status: 404 });
  }
  const url = new URL(`https://apollo.example${pathname}`);
  if (token !== null) {
    url.searchParams.set('token', token);
  }
  const authDecision = await authorizeMobileDeviceRequest({
    request: new Request(url.toString(), { headers: { Upgrade: 'websocket' } }),
    environment: ENVIRONMENT,
  });
  return authDecision.isAuthorized ? undefined : authDecision.response;
}

function connectAuthorized(
  endpoint: MobileDeviceEndpoint,
  connectionId: string,
): FakeWebSocket {
  const socket = new FakeWebSocket();
  endpoint.handleConnect({
    connectionId,
    port: socket,
    proof: buildAuthenticatedDeviceProof('device'),
  });
  return socket;
}

function listSourceFilePathList(): string[] {
  const filePathList: string[] = [];
  const walk = (directoryPath: string): void => {
    for (const entryName of readdirSync(directoryPath)) {
      const entryPath = join(directoryPath, entryName);
      if (statSync(entryPath).isDirectory()) {
        walk(entryPath);
        continue;
      }
      if (entryName.endsWith('.ts') && !entryName.endsWith('.spec.ts')) {
        filePathList.push(entryPath);
      }
    }
  };
  walk(SOURCE_ROOT);
  return filePathList;
}

function stripComment(source: string): string {
  return source.replace(/\/\*[\s\S]*?\*\//g, '').replace(/(^|[^:])\/\/.*$/gm, '$1');
}

function toRelativePath(filePath: string): string {
  return filePath.replace(SOURCE_ROOT, '').replace(/\\/g, '/').replace(/^\//, '');
}

describe('B7 Â§7+Â§9: structural guards', () => {
  it('10. the registry never imports execution or handlers', () => {
    const registrySource = stripComment(
      readFileSync(join(SOURCE_ROOT, 'devices/registry.ts'), 'utf8'),
    );
    for (const forbidden of [
      '@/tools/execution',
      '@/tools/router',
      '@/tools/types',

      'executeAuthorizedTool',
      'authorizeToolExecution',
    ]) {
      expect(registrySource).not.toContain(forbidden);
    }
  });

  it('12. target resolution does not depend on PcBridge', () => {
    // devices/target.ts es el resolver device-agnostic: no conoce transports.
    const targetSource = stripComment(
      readFileSync(join(SOURCE_ROOT, 'devices/target.ts'), 'utf8'),
    );
    expect(targetSource).not.toContain('getAgentByName');
    expect(targetSource).not.toContain('PcBridge');
    expect(targetSource).not.toContain('@/tools/execution');

    // Toda la capa de resoluciÃ³n (registry + target) es device-agnostic.
    for (const moduleName of ['devices/registry.ts', 'devices/target.ts']) {
      const moduleSource = stripComment(
        readFileSync(join(SOURCE_ROOT, moduleName), 'utf8'),
      );
      expect(moduleSource).not.toContain('getAgentByName');
      expect(moduleSource).not.toContain('@/devices/desktop');
      expect(moduleSource).not.toContain('@/tools/execution');
    }
  });

  it('the PcBridge transport lives only in the desktop device module', () => {
    // Dentro de devices/, el transporte PcBridge sÃ³lo puede existir en el mÃ³dulo
    // del deviceType que lo implementa (hoy: desktop).
    const pcBridgeModuleList = listSourceFilePathList()
      .filter((filePath) => toRelativePath(filePath).startsWith('devices/'))
      .filter((filePath) =>
        stripComment(readFileSync(filePath, 'utf8')).includes('getAgentByName'),
      );
    expect(pcBridgeModuleList.map(toRelativePath)).toEqual(['devices/desktop.ts']);
  });

  it('hostname is not a global identity source', () => {
    // El hostname sÃ³lo puede aparecer como identidad legacy del desktop.
    const hostnameModuleList = listSourceFilePathList().filter((filePath) =>
      stripComment(readFileSync(filePath, 'utf8')).includes(
        'buildDesktopDeviceIdFromHostname',
      ),
    );
    expect(hostnameModuleList.map(toRelativePath)).toEqual(['devices/desktop.ts']);
  });

  it('unimplemented capabilities can never become resolvable', () => {
    // Mobile: todas implementadas en Block 11.
    for (const capabilityName of MOBILE_CAPABILITY_NAME_LIST) {
      expect(isImplementedCapability(capabilityName)).toBe(true);

    }
  });
