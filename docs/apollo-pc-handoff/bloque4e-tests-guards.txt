function listSourceFilePathList(): string[] {
  const filePathList: string[] = [];
  const walk = (directoryPath: string): void => {
    for (const entryName of readdirSync(directoryPath)) {
      const entryPath = join(directoryPath, entryName);
      if (statSync(entryPath).isDirectory()) {
        walk(entryPath);
        continue;
      }
      if (entryName.endsWith('.ts') && !entryName.endsWith('.spec.ts')) {
        filePathList.push(entryPath);
      }
    }
  };
  walk(SOURCE_ROOT);
  return filePathList;
}

function stripComment(source: string): string {
  return source.replace(/\/\*[\s\S]*?\*\//g, '').replace(/(^|[^:])\/\/.*$/gm, '$1');
}

/**
 * El cuerpo de una funciÃ³n o mÃ©todo, desde su firma hasta la llave que la cierra.
 * Lo usan los guards que necesitan afirmar sobre UNA parte del archivo: cuando
 * un mÃ³dulo gana rutas legÃ­timas que no toca el camino que se quiere proteger,
 * un `not.toContain` sobre el archivo entero empieza a mentir.
 */
function readFunctionBody(source: string, signatureFragment: string): string {
  // Un fragmento puede aparecer primero como LUGAR DE LLAMADA y despuÃ©s como
  // declaraciÃ³n (`this.#buildHelloAuthorizer(auth);` vs `#buildHelloAuthorizer(
  // auth: ...)`). La llamada termina en `;` y no tiene cuerpo, asÃ­ que se prueban
  // todas las apariciones y se toma la primera que tenga cuerpo real.
  let searchFrom = 0;
  while (searchFrom < source.length) {
    const signatureIndex = source.indexOf(signatureFragment, searchFrom);
    if (signatureIndex === -1) {
      break;
    }
    const bodyBraceIndex = findBodyBraceIndex(source, signatureIndex);
    if (bodyBraceIndex !== -1) {
      return readBraceBlock(source, bodyBraceIndex, signatureFragment);
    }
    searchFrom = signatureIndex + signatureFragment.length;
  }
  throw new Error(`no se encontro el cuerpo de ${signatureFragment}`);
}

/**
 * La llave del CUERPO, no la de un tipo. Hay que saltar la lista de parÃ¡metros:
 * un input tipado como `{ readonly request: Request }` abre una llave que no es
 * la del cuerpo y el balance arrancarÃ­a desde el lugar equivocado.
 */
function findBodyBraceIndex(source: string, signatureIndex: number): number {
  let parenDepth = 0;
  for (let index = signatureIndex; index < source.length; index += 1) {
    const character = source[index];
    if (character === '(') {
      parenDepth += 1;
      continue;
    }
    if (character === ')') {
      parenDepth -= 1;
      continue;
    }
    if (parenDepth !== 0) {
      continue;
    }
    if (character === '{') {
      return index;
    }
    if (character === ';') {
      return -1;
    }
  }
  return -1;
}

function readBraceBlock(
  source: string,
  openBraceIndex: number,
  signatureFragment: string,
): string {
  let depth = 0;
  for (let index = openBraceIndex; index < source.length; index += 1) {
    if (source[index] === '{') {
      depth += 1;
      continue;
    }
    if (source[index] !== '}') {
      continue;
    }
    depth -= 1;
    if (depth === 0) {
      return source.slice(openBraceIndex, index + 1);
    }
  }
  throw new Error(`el cuerpo de ${signatureFragment} no cierra`);
}

  const filePathList: string[] = [];

    for (const entryName of readdirSync(directoryPath)) {
      const entryPath = join(directoryPath, entryName);
      if (statSync(entryPath).isDirectory()) {
        walk(entryPath);
        continue;
      }
      if (entryName.endsWith('.ts') && !entryName.endsWith('.spec.ts')) {
        filePathList.push(entryPath);
      }
    }
  };
  walk(SOURCE_ROOT);
  return filePathList;
}

function toRelativePath(filePath: string): string {
  return filePath.replace(SOURCE_ROOT, '').replace(/\\/g, '/').replace(/^\//, '');
}

function readModuleSource(moduleRelativePath: string): string {
  return readFileSync(join(SOURCE_ROOT, moduleRelativePath), 'utf8');
}

function listImportPathList(source: string): string[] {
  return [...source.matchAll(/from '(@\/[^']+)'/g)].map((match) => match[1] ?? '');
}

describe('B6 security: the registry cannot execute or authorize', () => {
  it('12. no device module imports authorization or execution', () => {
    const forbiddenImportList = ['@/tools/execution', '@/tools/router', '@/tools/types'];
    const offenderList = listSourceFilePathList()
      .filter((filePath) => filePath.replace(/\\/g, '/').includes('/devices/'))
      .filter((filePath) => {
        const importPathList = listImportPathList(readFileSync(filePath, 'utf8'));
        return importPathList.some((importPath) =>
          forbiddenImportList.includes(importPath),
        );
      });
    expect(offenderList.map(toRelativePath)).toEqual([]);
  });

  it('12b. no device module invokes a handler or a tool executor', () => {
    const offenderList = listSourceFilePathList()
      .filter((filePath) => filePath.replace(/\\/g, '/').includes('/devices/'))
      .filter((filePath) => {
        const source = readFileSync(filePath, 'utf8');

          /\.handler\s*\(/.test(source) ||
          source.includes('executeToolByName(') ||
          source.includes('resolveToolExecution(') ||
          source.includes('authorizeToolExecution(')
        );
      });

  });

  it('12c. execution.ts is still the only module with a handler call', () => {
