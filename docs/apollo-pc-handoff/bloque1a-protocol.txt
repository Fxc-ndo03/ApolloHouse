  readonly requestId: string;
  readonly clientTurnId?: string;
  readonly generation?: number;
  readonly capability: CapabilityName;
  // BLOCK 11 Â§4: comando explicito de la capability. Opcional: si falta, el peer
  // usa el canÃ³nico (el primero del commandList), que es lo que mandaba un peer
  // de Block 9.
  readonly command?: string;
  readonly targetDeviceId: string;
  readonly payload?: Readonly<Record<string, unknown>>;
}): DeviceCommandEnvelope {
  return deviceCommandEnvelopeSchema.parse({
    requestId: input.requestId,
    clientTurnId: input.clientTurnId,
    generation: input.generation,
    capability: input.capability,
    command: input.command,
    targetDeviceId: input.targetDeviceId,
    payload: input.payload ?? {},
  });
}
